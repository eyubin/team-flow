function csrfToken() {
  return document.cookie
    .split('; ')
    .find((cookie) => cookie.startsWith(`${csrfCookie}=`))
    ?.split('=')[1]
}

async function signOut() {
  await fetch('/api/auth/csrf', { credentials: 'include' })
